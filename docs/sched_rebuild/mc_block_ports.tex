\begin{center}
{\LARGE\textbf{MC-Core Block \& Port Reference}}\\[2pt]
{\small One channel, AXI $\rightarrow$ DFI. Every sub-block: ports (dir/width), valid-ready,
and \inst{instance count}. Params: \texttt{N\_RANKS}(=2), \texttt{N\_BANKS}=32 (8 BG$\times$4),
\texttt{N\_BG}=8, \texttt{BL}=16, \texttt{GEAR}$\in\{1,2,4\}$, \texttt{ROB\_IDX}=8b, 64B-aligned addr.}
\end{center}

\section{Boundary interfaces}
\subsection{AXI write channels \inst{$\times$1}}
\begin{ports}
awid & in & AXI\_ID\_W & write addr id\\
awaddr & in & ADDR\_W & 64B-aligned (LSB[5:0]=0)\\
awlen/awsize/awburst & in & 8/3/2 & beats-1 / bytes-per-beat / burst type\\
awvalid/awready & i/o & 1/1 & AW handshake\\
wdata/wstrb & in & 512/64 & write beat + byte enables (DM)\\
wlast/wvalid/wready & i/o & 1 & W handshake; wlast = last beat\\
bid/bresp/bvalid/bready & o/o/o/i & ID/2/1/1 & write response (self-timed)\\
\end{ports}
\subsection{AXI read channels \inst{$\times$1}}
\begin{ports}
arid/araddr/arlen/arsize/arburst & in & ID/ADDR/8/3/2 & read request\\
arvalid/arready & i/o & 1 & AR handshake\\
rid/rdata/rresp/rlast & out & ID/512/2/1 & read return beats\\
rvalid/rready & o/i & 1 & R handshake\\
\end{ports}
\subsection{DFI egress \inst{$\times$1} (to PHY)}
\begin{ports}
dfi\_address & out & 14$\times$GEAR & CA bus, per phase (2-cyc cmds)\\
dfi\_cs\_n/dfi\_act\_n & out & (2/1)$\times$GEAR & chip-select / activate per phase\\
dfi\_bg/dfi\_bank & out & (3/2)$\times$GEAR & bank group / bank\\
dfi\_wrdata/\_mask/\_en & out & (2$\cdot$32/16/1)$\times$GEAR & write data + DM + enable\\
dfi\_rddata/\_valid & in & (2$\cdot$32/1)$\times$GEAR & read data + valid (RL later)\\
dfi\_cke/dfi\_odt/dfi\_reset\_n & out & N\_RANKS & power / ODT / reset\\
dfi\_freq\_ratio/\_init\_start & out & 2/1 & gear + freq-change handshake\\
dfi\_init\_complete/dfi\_alert\_n & in & 1/1 & PHY ready / write-CRC alert\\
\end{ports}

\section{Request admit + write ROB \inst{$\times$1}}

\subsection{write\_rob \inst{$\times$1} (depth ROB\_DEPTH)}
\begin{ports}
alloc\_in/alloc\_we & in & --/1 & \{aw\_id, addr, narrow\}; admit iff !full\\

\subsection{tcam \{aw\_id\} \inst{$\times$1} (depth ROB\_DEPTH)}

\end{ports}

\subsection{cmd\_arb \inst{$\times$64}}

(stub) + Error (stub). Full port lists in \texttt{fig\_39--46}. Common bus:
\begin{ports}
me\_cmd\_valid/type/rank/bg/bank/mr & out & 1/3/../.. & flag to scheduler\\
sched\_ack & in & 1 & packer accepted\\
set/clr\_gate\_rfc/zq/mr & out & N\_RANKS & scoreboard gates\\
raa\_inc\_en/raa\_dec & i/o & 16$\times$N\_RANKS & RFM counters\\
mrr\_data\_valid/data/rank/requester & in & 1/8/../.. & RD\_CAP sideband\\
init\_done & out & 1 & DFI mux handoff\\
\end{ports}
