{\LARGE\textbf{DDR command-bus truth tables --- DDR1 / DDR2 / DDR3 / DDR4}}\\[2pt]
{\small Command ports only. \textbf{Structural difference vs DDR5:} DDR1--4 decode the command from
\emph{dedicated} pins (\texttt{CS\_n, RAS\_n, CAS\_n, WE\_n}; DDR4 adds \texttt{ACT\_n}) with bank/row/col
on \emph{separate} address pins --- there is no encoded 2-cycle CA bus. All commands are \textbf{1 CK}
(SDR command). DDR5 = the encoded CA bus (see \texttt{opcode\_rom.pdf}).
\colorbox{opc}{green}=fixed command bit; \colorbox{fld}{yellow}=address field. H=1, L=0, x=don't care.
\emph{Verified against JESD79F (DDR1 TT-3), JESD79-2F (Table 13), JESD79-3F (p47 cmd TT),
JESD79-4 (Table on p32).}}
\end{center}

\section{DDR1 / DDR2 / DDR3  (shared RAS/CAS/WE encoding)}

\begin{tabular}{F C C C C K C M}

\rowcolor{hb}\textbf{command} & \textbf{CS\_n} & \textbf{RAS\_n} & \textbf{CAS\_n} & \textbf{WE\_n} & \textbf{BA} & \textbf{A10} & \textbf{note}\\ \midrule
Deselect (DES) & \op{H} & x & x & x & x & x & CS high: no command latched.\\
No-Op (NOP) & \op{L} & \op{H} & \op{H} & \op{H} & x & x & idle, holds state.\\
Activate (ACT) & \op{L} & \op{L} & \op{H} & \op{H} & \fd{bank} & \fd{row} & opens row; row on the A pins, bank on BA.\\
Read (RD) & \op{L} & \op{H} & \op{L} & \op{H} & \fd{bank} & \fd{AP} & column on A; A10 = auto-precharge.\\
Write (WR) & \op{L} & \op{H} & \op{L} & \op{L} & \fd{bank} & \fd{AP} & column on A; A10 = auto-precharge.\\
Precharge (PRE) & \op{L} & \op{L} & \op{H} & \op{L} & \fd{bank} & \fd{all/one} & A10=1 all banks, A10=0 the one BA.\\
Auto-Refresh (REF) & \op{L} & \op{L} & \op{L} & \op{H} & x & x & CKE high both edges.\\
Self-Refresh entry (SRE) & \op{L} & \op{L} & \op{L} & \op{H} & x & x & same code as REF but CKE H$\rightarrow$L.\\

ZQ Cal (DDR3 only) & \op{L} & \op{H} & \op{H} & \op{L} & x & \fd{long/short} & ZQCL (A10=1) / ZQCS (A10=0).\\
Power-Down entry (PDE) & \op{L} & \op{H} & \op{H} & \op{H} & x & x & NOP/DES with CKE H$\rightarrow$L.\\
PD / SR exit (PDX/SRX) & \multicolumn{7}{l}{\footnotesize CKE L$\rightarrow$H (command pins x); DRAM resumes after tXP / tXS.}\\
\bottomrule
\end{tabular}}}
\smallskip
{\footnotesize DDR1: 4 banks (BA0--1), no ODT, no reset\_n, no ZQ. DDR2: adds ODT, up to 8 banks (BA0--2),
extra EMRs. DDR3: 8 banks, adds \texttt{reset\_n}, ZQ calibration, dynamic ODT, write-leveling. Command
decode is identical across the three --- only the surrounding pins/features grow.}

\section{DDR4  (adds ACT\_n; RAS/CAS/WE repurposed as A16/A15/A14)}
{\footnotesize When \texttt{ACT\_n}=L the command is ACTIVATE and the RAS/CAS/WE pins carry the high row
bits A16/A15/A14. When \texttt{ACT\_n}=H those same pins decode every other command (as RAS\_n/CAS\_n/WE\_n).}

Deselect (DES) & \op{H} & x & x & x & x & x & x\\
Activate (ACT) & \op{L} & \op{L} & \fd{A16} & \fd{A15} & \fd{A14} & \fd{grp,bank} & \fd{row}\\
No-Op (NOP) & \op{L} & \op{H} & \op{H} & \op{H} & \op{H} & x & x\\

\bottomrule
\end{tabular}}
\smallskip
{\footnotesize \textbf{notes:} DES = CS high (no decode). ACT (ACT\_n=L): RAS/CAS/WE carry row bits
A16/A15/A14; BG,BA pick the bank. RD/WR: column on the A pins, A10 = auto-precharge. PRE: A10=1 all
banks / 0 the one BG,BA. MRS: BG,BA select MR0--MR6, A = opcode. ZQ: A10 = long/short. SRE/PDE =
REF-code / NOP with CKE H$\rightarrow$L; PDX/SRX = CKE L$\rightarrow$H.}
\smallskip
{\footnotesize DDR4: up to 16 banks = 4 bank groups (BG0--1) $\times$ 4 banks (BA0--1); adds \texttt{ACT\_n},
\texttt{PAR} (CA parity), CID (3DS stacking), DBI, CA gear-down, tCCD\_L/S. Command still 1 CK.}
