\begin{center}
{\LARGE\textbf{Scheduler --- Black-box I/O}}\\[2pt]
{\small One channel, DDR5. The scheduler = the final select/pack bar. The per-rank ARBITERS
(rank\_act\_abr, rank\_cas\_abr, pick-2-pre) sit UPSTREAM and hand it the winners; the scheduler
packs the bundle and drives the phaser. \textcolor{ingr}{in} = into scheduler, \textcolor{outg}{out}
= from scheduler.}
\end{center}

\section{INPUTS  (into the scheduler)}
\subsection{Per-rank arb winners  (from rank\_act\_abr / rank\_cas\_abr / pick-2-pre)}
\begin{io}
act\_win[r] = \{rdy, bg, bank\} & few $\times$N\_RANKS & \textcolor{ingr}{in} & the 1 ACT winner / rank (already \& can\_act\_rank\_s upstream)\\
act\_row[r] & row $\times$N\_RANKS & \textcolor{ingr}{in} & row for that ACT (row\_sel)\\
cas\_win[r] = \{rdy, bg, bank\} & few $\times$N\_RANKS & \textcolor{ingr}{in} & the 1 CAS winner / rank (\& can\_cas\_rank\_s upstream)\\
cas\_col[r] = \{idx, wr/rd\_ptr, column\} & bus $\times$N\_RANKS & \textcolor{ingr}{in} & col + sram ptr + rob for that CAS (col\_sel)\\
pre\_win[r][0..1] = \{rdy, bg, bank\} & few$\times$2$\times$N\_RANKS & \textcolor{ingr}{in} & the 2 PRE winners / rank (pick-2-pre off p\_bit)\\
\end{io}
\subsection{Global / cross-rank legality  (for the L3 CAS pick + packing)}
\begin{io}
can\_cas\_dq & 1 & \textcolor{ingr}{in} & DQ bus free (dqFree=BL/2); only \emph{one} CAS may win the bundle\\
can\_wr / can\_rd & 1 ea & \textcolor{ingr}{in} & R<->W turnaround gap (tWTR/tRTW) on a direction flip\\
last\_cas\_dir/rank/bg & few & \textcolor{ingr}{in} & bus state -> cross-rank tRTRS, diff-BG packing, dir bias\\
batch\_dir[r] & 1 $\times$N\_RANKS & \textcolor{ingr}{in} & WLR R/W batch direction (rank bias at L3)\\
gear & 2 & \textcolor{ingr}{in} & 1/2/4 -> number of phase slots to fill\\
\end{io}
\subsection{Maintenance flags  (ME -> scheduler, ranked with demand)}
\begin{io}
refsb\_rdy / ref\_force & 1 ea & \textcolor{ingr}{in} & refresh owed (|debt) / forced (\&debt==7)\\
rfm\_rdy / rfm\_force & 1 ea & \textcolor{ingr}{in} & rowhammer owed (RAA>=IMT) / hard (>=MMT)\\
zq\_rdy & 1 & \textcolor{ingr}{in} & ZQ calibration due\\
mrr\_rdy & 1 & \textcolor{ingr}{in} & MR-read pending (temp poll / read MR) -- uses DQ\\
mrw\_rdy & 1 & \textcolor{ingr}{in} & MR-write pending (config / freq-change)\\
cas\_complete[b] & 1 $\times$N\_BANKS & \textcolor{ingr}{in} & bank drained + precharged (state==IDLE) -- issue gate\\
maint\_cmd\_type & 5 & \textcolor{ingr}{in} & MAINT\_PRIO winner opcode (REF/RFM/ZQ/MRW/MRR)\\
init\_done & 1 & \textcolor{ingr}{in} & boot done (else DFI mux on init, scheduler gated)\\
dfi\_alert\_n / dfi\_error & 1 ea & \textcolor{ingr}{in} & PHY alert / error\\
\end{io}

\section{OUTPUTS  (from the scheduler)}
\subsection{Bundle -> cmd\_phaser}
\begin{io}
CMD\_0..3\_PACKET (vld + record) & $\sim$49$\times$4 & \textcolor{outg}{out} & \{valid, cmd\_type, rw, rank, bg, bank, addr, ap, bl, rob\_index, sram\_addr\} per phase slot (1/2/4 used by gear)\\
two\_ui / repeat & 1 & \textcolor{outg}{out} & side wire -> DFI driver (1-UI vs 2-UI; @1:1 hold)\\
\end{io}
\subsection{Grants / commit  (back to cells + scoreboard)}
\begin{io}
grant\_act\_oh / grant\_cas\_oh & 1 ea $\times$N\_BANKS & \textcolor{outg}{out} & the chosen ACT/CAS winner one-hot; \textbf{pop-on-CAS} retires packet\\
grant\_pre\_oh[0..1] & 1$\times$2 $\times$N\_BANKS & \textcolor{outg}{out} & the chosen PRE winners one-hot\\
fsm\_commit (issue-edge) & pulse$\times$class & \textcolor{outg}{out} & sole commit: updates BANK\_FSM + fires the timestamp writeback\\
cmd\_issued\{r,bg,bank,cmd\} & bus & \textcolor{outg}{out} & drives next\_* writeback (WE + const-router select)\\
\end{io}
\subsection{Maintenance handshake  (to ME)}
\begin{io}
maint\_ack & 1 & \textcolor{outg}{out} & the maint cmd (cmd\_type) issued -> ME clears debt/RAA, advances ba\\
\end{io}

\vspace{4pt}
\section*{Upstream (feeds the winners --- NOT scheduler I/O)}
{\footnotesize The per-rank arbiters are a SEPARATE stage before the scheduler. Their I/O:
\textbf{in} = PRE\_BANK\_ROW/COL\_PATHS$\times$32 (per-bank need/can\_act/cas/pre, is\_hit, age,
open\_row, col/ptr/rob), \texttt{p\_bit} (PRE-elig vector), rank-scope gates
\texttt{can\_act\_rank\_s}/\texttt{can\_cas\_rank\_s}, BG gates. \textbf{out} = act\_win/cas\_win/
pre\_win (the scheduler's inputs above). So \texttt{can\_*}, \texttt{p\_bit}, weights (CAS\_W/ACT\_W/
PRE\_W) are consumed HERE, not in the scheduler bar.}

\vspace{3pt}
{\footnotesize \textbf{Scheduler bar's job:} take the per-rank winners + maint flags, apply the
cross-rank / global constraints (one CAS via dqFree, turnaround, tRTRS, dir bias), rank maint WITH
demand (MAINT\_PRIO), and PACK the CMD\_0..3 bundle for the phaser --- plus feed back grant\_oh
(pop-on-CAS) and the fsm\_commit writeback edge. It never rejects: legality was folded into the
winners' \texttt{rdy} bits upstream.}
