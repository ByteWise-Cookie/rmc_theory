\begin{center}
{\LARGE\textbf{Scheduler --- Black-box I/O}  \normalsize (matches arb drawing)}\\[2pt]
{\small One channel, DDR5. Per rank per cycle the arb emits \textbf{1 ACT + 1 CAS + 2 PRE} winners,
each \{vld,bg,bank\}, gated by the rank-scope can. Row path -> ACT, col path -> CAS. Parametric
N\_BANKS(=N\_BG$\times$4), N\_BG, N\_RANKS.}
\end{center}

\section{INPUTS}
\subsection{Per-bank path candidates  (PRE\_BANK\_ROW\_PATHS / COL\_PATHS $\times$32)}
\begin{io}
can\_act[b] / can\_cas[b] / can\_pre[b] & 1 ea $\times$N\_BANKS & \textcolor{ingr}{in} & per-bank legality (into row/col/pre paths)\\
need\_act / need\_cas / need\_pre[b] & 1 ea $\times$N\_BANKS & \textcolor{ingr}{in} & resident packet demand\\
p\_bit[b] & 1 $\times$N\_BANKS & \textcolor{ingr}{in} & PRE-eligible vector (need\_pre \& can\_pre) -> pick-2-pre\\
is\_hit[b] / age[b] & 1/few $\times$N\_BANKS & \textcolor{ingr}{in} & CAS\_W row-lock / ACT\_W age (weights)\\
open\_row[b] / col,rd\_wr\_ptr,rob & bus $\times$N\_BANKS & \textcolor{ingr}{in} & row (ACT path) / \{col, sram ptr, rob\} (CAS path) from BANK\_REQ\\
\end{io}
\subsection{Rank-scope gates  (AND'd with the class winner)}
\begin{io}
can\_act\_rank\_s[r] & 1 $\times$N\_RANKS & \textcolor{ingr}{in} & rank ACT gate: tRRD\_S (cross-BG) + tFAW window\\
can\_cas\_rank\_s[r] & 1 $\times$N\_RANKS & \textcolor{ingr}{in} & rank CAS gate: turnaround (tWTR/tRTW) + dqFree\\
can\_act\_bgs / can\_cas\_bgs[g] & 1 ea $\times$N\_BG & \textcolor{ingr}{in} & same-BG spacing (tRRD\_L / tCCD\_L(\_WR))\\
\end{io}
\subsection{Maintenance flags  (ME -> scheduler, ranked with demand)}
\begin{io}
refsb\_rdy / ref\_force & 1 ea & \textcolor{ingr}{in} & refresh owed (|debt) / forced (\&debt==7)\\
rfm\_rdy / rfm\_force & 1 ea & \textcolor{ingr}{in} & rowhammer owed (RAA>=IMT) / hard (>=MMT)\\
zq\_rdy & 1 & \textcolor{ingr}{in} & ZQ calibration due (tZQ)\\
mrr\_rdy & 1 & \textcolor{ingr}{in} & MR-read pending (temp poll / read MR) -- uses DQ (dqFree)\\
mrw\_rdy & 1 & \textcolor{ingr}{in} & MR-write pending (config / freq-change)\\
cas\_complete[b] & 1 $\times$N\_BANKS & \textcolor{ingr}{in} & bank drained + precharged (state==IDLE)\\
maint\_cmd\_type & 5 & \textcolor{ingr}{in} & MAINT\_PRIO winner opcode (REF/RFM/ZQ/MRW/MRR)\\
init\_done & 1 & \textcolor{ingr}{in} & boot done (else DFI mux on init, scheduler gated)\\
\end{io}
\subsection{Control / bus state}
\begin{io}
gear & 2 & \textcolor{ingr}{in} & mc:CK ratio -> bundle width (dfi\_cmd $\times$ gear)\\
batch\_dir[r] & 1 $\times$N\_RANKS & \textcolor{ingr}{in} & WLR R/W batch direction (dir bias)\\
op\_code (cmd) & few & \textcolor{ingr}{in} & feedback\_ctl -> const-router select (writeback)\\
last\_cas\_dir/rank/bg & few & \textcolor{ingr}{in} & bus state (turnaround + diff-BG packing)\\
dfi\_alert\_n / dfi\_error & 1 ea & \textcolor{ingr}{in} & PHY alert / error\\
\end{io}

\section{OUTPUTS}
\subsection{Per-rank class winners  (the arb result)}
\begin{io}
act\_win[r] = \{vld, bg, bank\} & few $\times$N\_RANKS & \textcolor{outg}{out} & \textbf{1 ACT winner / rank} (rank\_act\_abr \& can\_act\_rank\_s)\\
act\_row[r] & row $\times$N\_RANKS & \textcolor{outg}{out} & row for the ACT winner (row\_sel of the winning bank)\\
cas\_win[r] = \{vld, bg, bank\} & few $\times$N\_RANKS & \textcolor{outg}{out} & \textbf{1 CAS winner / rank} (rank\_cas\_abr \& can\_cas\_rank\_s)\\
cas\_col[r] = \{idx, wr/rd\_ptr, column\} & bus $\times$N\_RANKS & \textcolor{outg}{out} & col + sram ptr + rob for the CAS winner (col\_sel)\\
pre\_win[r][0..1] = \{vld, bg, bank\} & few$\times$2$\times$N\_RANKS & \textcolor{outg}{out} & \textbf{2 PRE winners / rank} (pick-2-pre from p\_bit)\\
\end{io}
\subsection{Bundle to cmd\_phaser}
\begin{io}
CMD\_0..3\_PACKET (vld + record) & $\sim$49$\times$4 & \textcolor{outg}{out} & \{valid, cmd\_type, rw, rank, bg, bank, addr, ap, bl, rob\_index, sram\_addr\} per phase slot\\
two\_ui / repeat & 1 & \textcolor{outg}{out} & side wire -> DFI driver (1-UI vs 2-UI; @1:1 hold)\\
\end{io}
\subsection{Commit / feedback}
\begin{io}
grant\_act\_oh / grant\_cas\_oh & 1 ea $\times$N\_BANKS & \textcolor{outg}{out} & winner one-hot -> cells; \textbf{pop-on-CAS} retires the packet\\
grant\_pre\_oh[0..1] & 1$\times$2 $\times$N\_BANKS & \textcolor{outg}{out} & the 2 PRE winners one-hot\\
fsm\_commit (issue-edge) & pulse$\times$class & \textcolor{outg}{out} & sole commit: BANK\_FSM state + fires timestamp writeback\\
cmd\_issued\{r,bg,bank,cmd\} & bus & \textcolor{outg}{out} & drives next\_* writeback (WE + const-router)\\
\end{io}
\subsection{Maintenance handshake  (to ME)}
\begin{io}
maint\_ack & 1 & \textcolor{outg}{out} & the maint cmd (cmd\_type) issued -> ME clears debt/RAA, advances ba\\
\end{io}

\vspace{3pt}
{\footnotesize \textbf{Structure (per drawing):} each rank has \texttt{rank\_act\_abr} (-> 1 ACT winner)
and \texttt{rank\_cas\_abr} (-> 1 CAS winner), each ANDed with its rank-scope can (\texttt{can\_act\_rank\_s}
/ \texttt{can\_cas\_rank\_s}); PRE\_PATHS \texttt{pick-2-pre} off the \texttt{p\_bit} vector (-> 2 PRE
winners, tPPD allows 2). ACT pulls its row via \texttt{row\_sel}; CAS pulls col+ptr via \texttt{col\_sel}.
Winners + maint (ranked by MAINT\_PRIO) pack into CMD\_0..3 for the phaser (1:1/1:2/1:4 = 1/2/4 slots).
The arb NEVER rejects --- legality is folded into the \texttt{can\_*} inputs; it only selects + weights.}
