\section{Gate model (recap)}
Every command a bank can issue is legal only when its \textbf{4 folded 1-bit gates} are all
high. Each timing parameter is owned by exactly one \emph{scope level} and drives one gate at
that level; the resident gate the eligibility logic reads is the AND across levels:
\[
\p{can\_x} \;=\; \p{can\_x\_L0}\;\&\;\p{can\_x\_L1}\;\&\;\p{can\_x\_L2}\;\&\;\p{can\_x\_L3},
\qquad x\in\{\p{act},\p{pre},\p{rcas},\p{wcas}\}.
\]
Each level bit is a comparator $\p{can\_x\_Ln} = \big(\text{GC} \ge \text{next\_x\_Ln}\big)$,
where $\text{next\_x} = \text{GC}_{\text{issue}} + t$. Scopes:

\begin{center}\small
\begin{tabular}{@{}l l l l@{}}
\toprule
\textbf{Level} & \textbf{scope} & \textbf{instances} & \textbf{tracks} \\
\midrule
\rowcolor{l0}\p{L0} & same bank            & $N_{rank}\!\times\!32=64$ & this bank's own last op \\
\rowcolor{l1}\p{L1} & same bank-group      & $N_{rank}\!\times\!8=16$  & any sibling bank in the BG \\
\rowcolor{l2}\p{L2} & diff-BG, same rank   & $N_{rank}=2$              & rank-wide last op + tFAW window \\
\rowcolor{l3}\p{L3} & diff rank            & $1$ (global)              & DQ-bus rank handoff \\
\bottomrule
\end{tabular}
\end{center}

\section{Per-level gate equations}
\rowcolors{1}{}{}

\subsection{L0 — same bank (per cell $\times64$)}
\begin{align*}
\p{can\_act\_L0}  &= (\text{GC}\ge \text{lastPRE}+t_{RP}) \,\&\, (\text{GC}\ge \text{lastACT}+t_{RC})\\
\p{can\_pre\_L0}  &= (\text{GC}\ge \text{lastACT}+t_{RAS}) \,\&\, (\text{GC}\ge \text{lastRD}+t_{RTP}) \,\&\, (\text{GC}\ge \text{lastWR}+t_{WR})\\
\p{can\_rcas\_L0} &= (\text{GC}\ge \text{lastACT}+t_{RCD}) \,\&\, (\text{GC}\ge \text{lastCAS}+t_{CCD\_L}) \,\&\, (\text{GC}\ge \text{lastWR}+t_{WTR\_L})\\
\p{can\_wcas\_L0} &= (\text{GC}\ge \text{lastACT}+t_{RCD}) \,\&\, (\text{GC}\ge \text{lastCAS}+t_{CCD\_L\_WR}) \,\&\, (\text{GC}\ge \text{lastRD}+t_{RTW})
\end{align*}

\subsection{L1 — same bank-group (per BG $\times16$, broadcast to its 4 banks)}
\begin{align*}
\p{can\_act\_L1}  &= \text{GC}\ge \text{lastACT}_{bg}+t_{RRD\_L}\\
\p{can\_pre\_L1}  &= 1 \quad(\text{no same-BG PRE rule})\\
\p{can\_rcas\_L1} &= (\text{GC}\ge \text{lastRD}_{bg}+t_{CCD\_L}) \,\&\, (\text{GC}\ge \text{lastWR}_{bg}+t_{WTR\_L})\\
\p{can\_wcas\_L1} &= \text{GC}\ge \text{lastWR}_{bg}+t_{CCD\_L\_WR}
\end{align*}

\subsection{L2 — diff-BG, same rank (per rank $\times2$)}
\begin{align*}
\p{can\_act\_L2}  &= (\text{GC}\ge \text{lastACT}_{rk}+t_{RRD\_S}) \,\&\, (\text{GC}\ge \text{faw\_ts[oldest]}+t_{FAW})\\
\p{can\_pre\_L2}  &= \text{GC}\ge \text{lastPRE}_{rk}+t_{PPD}\\
\p{can\_rcas\_L2} &= (\text{GC}\ge \text{lastCAS}_{rk}+t_{CCD\_S}) \,\&\, (\text{GC}\ge \text{lastWR}_{rk}+t_{WTR\_S})\\
\p{can\_wcas\_L2} &= (\text{GC}\ge \text{lastCAS}_{rk}+t_{CCD\_S}) \,\&\, (\text{GC}\ge \text{lastRD}_{rk}+t_{RTW})
\end{align*}

\subsection{L3 — diff rank (global $\times1$)}
\begin{align*}
\p{can\_act\_L3}  &= 1 \qquad \p{can\_pre\_L3} = 1 \quad(\text{cross-rank ACT/PRE free})\\
\p{can\_rcas\_L3} = \p{can\_wcas\_L3} &= \underbrace{(\text{GC}\ge \p{dqFree\_ts})}_{\text{DQ occupancy}} \,\&\, \underbrace{\big[(\text{rank}{=}\text{last\_cas\_rank})\,?\,1:(\text{GC}\ge \text{last\_cas\_ts}+t_{RTRS})\big]}_{\text{bus handoff}}
\end{align*}
\noindent where $\p{dqFree\_ts}=\text{last\_cas\_ts}+\text{BL}/2\,(8)$, written by the packer at
CAS-commit. \textbf{dqFree is the binding cross-rank floor:} same-rank CAS already pay
$t_{CCD\_S}=8\ge$ burst, but a rank hop pays only $t_{RTRS}=2<8$ — tRTRS covers the electrical
handoff (ODT/driver), \emph{not} the 8-CK burst occupancy, so without dqFree two ranks would dump
overlapping bursts on DQ. A \emph{fresh/idle} CAS has $\p{dqFree\_ts}$ in the past $\Rightarrow$
gate open, issue now. The RD$\leftrightarrow$WR \emph{direction-flip} hole is separate
($t_{WTR}$/$t_{RTW}$, already carried at L0--L2 rank scope, since there CL$\ne$CWL). The CA-bus
slot ($t_{caFree}$, 2-cyc $=$ 2 phases) and the data-engine-free check (WD\_SRAM fetch $+$
splitter/accumulator busy $=$ burst $=$ subsumed by dqFree) live at the \textbf{packer}, not as
selection gates.

\section{Datapath-busy timing}
The DQ bus is the real throughput limiter, not the command bus. One CAS $=$ one BL16 burst $=$
\textbf{8 CK on DQ}. Sustained ceiling: \textbf{$\le 1$ CAS per 8 CK} $= t_{CCD\_S}$. Back-to-back
diff-BG CAS at tCCD\_S$=8$ exactly fills the DQ with zero bubble (the DQ heartbeat); same-BG at
tCCD\_L$=12$ leaves a 4-CK gap; same-BG writes at tCCD\_L\_WR$=48$ leave a 40-CK crater.

\subsection{Read path (CAS to completion)}
\begin{center}\small
\begin{tabular}{@{}r l@{}}
\toprule
\textbf{CK from RD CAS} & \textbf{event} \\
\midrule
0    & RD CAS issued (col on CA bus) \\
40   & first read beat on DQ ($=$CL); DQ busy \p{40..47} \\
$\sim$48 & last beat captured; RD accumulator has full 64\,B \\
$\sim$48 & one wide 512-b write into \p{RD\_SRAM} @ \p{dbuf\_addr} \\
$\sim$49 & \p{rd\_done\{rob\_index,pkt\_num\}} $\rightarrow$ COMPLETION $\rightarrow$ resp AFIFO \\
\bottomrule
\end{tabular}
\end{center}
\p{RL\_line} is a deterministic CL-deep shift carrying only \p{\{rob\_index,dbuf\_addr\}}; it
self-pops at CL. No CAM. Completion never fires before the final beat is committed to SRAM.

\subsection{Write path (CAS to retire) --- two separate completion events}
\begin{center}\small
\begin{tabular}{@{}r l@{}}
\toprule
\textbf{CK from WR CAS} & \textbf{event} \\
\midrule
0    & WR CAS issued (col on CA bus) \\
38   & first write beat driven on DQ ($=$CWL); WD splitter reads 512\,b from \p{WD\_SRAM} \\
46   & last beat driven; DQ free \\
\textbf{46} & \textbf{CIF-facing done}: \p{wr\_done} $\rightarrow$ COMPLETION, \p{wd\_slot} freed (self-timed, no bus ack) \\
\textbf{118}& \textbf{bank retire}: $46+t_{WR}(72)$ --- earliest the bank may PRE \\
\bottomrule
\end{tabular}
\end{center}
\textbf{Two distinct events} (do not conflate): the CIF completion + slot free at \textbf{46}
(write is fire-and-forget, controller-generated, no DRAM response), and the array-level retire at
\textbf{118} (the cell capacitor finished charging). Precharging before 118 disconnects the sense
amps mid-restore $\Rightarrow$ data loss. This is why a same-bank write streak stacks tWR: each
different-row write pays the full close cycle.

\subsection{Physical write cycle (why same-bank writes cost 198)}
\p{WR $\rightarrow$ PRE} $=118$, $+\,t_{RP}(40)\rightarrow$ ACT @ 158, $+\,t_{RCD}(40)\rightarrow$
WR @ 198. So a same-bank \emph{different-row} write cycle $=198$ CK; a same-row write is just
$t_{CCD\_L\_WR}=48$ (no PRE, tWR paid once at the final close). Read cycle same-bank different-row
$= t_{RC}=117$.

\section{Datapath timing constraints (summary)}
\begin{itemize}[leftmargin=1.2em,itemsep=1pt]
\item \textbf{DQ ceiling:} $\le1$ CAS / 8 CK sustained. Extra command-bus phases fill with
      ACT/PRE/REF to \emph{other} banks.
\item \textbf{Read latency:} data lands CL$=40$ CK after RD CAS; RD accumulator does one 512-b
      SRAM write per packet (no per-beat dribble).
\item \textbf{Write latency:} data leaves CWL$=38$ CK after WR CAS; WR splitter does one 512-b
      SRAM read per packet.
\item \textbf{Write retire:} bank locked from PRE until CAS$+118$; CIF completion decoupled at
      CAS$+46$.
\item \textbf{Turnaround on one DQ bus:} R$\rightarrow$W $=$ tRTW$=12$; W$\rightarrow$R $=$ CWL$+8+$tWTR
      ($70$/$52$ same/diff-BG), skipped cross-rank ($\sim$tRTRS).
\item \textbf{RL/WL lines carry no timestamp} --- fixed shift depth \emph{is} the timing; load at
      CAS-commit, self-pop at CL/CWL.
\item \textbf{Gear:} at 1:$N$, $1$ mc\_clk $=N$ CK; GC counts CK and advances by $N$/mc\_clk so
      all deadlines stay CK-granular. Accumulator/splitter are the only gear-aware blocks.
\end{itemize}
